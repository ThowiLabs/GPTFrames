cmake_minimum_required(VERSION 3.16)
project(hyperframes_cpp_native LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(HFC_HTML_LAYOUT "Build the litehtml HTML/CSS video renderer" ON)
include(cmake/EmbedFonts.cmake)
include_directories("${CMAKE_CURRENT_BINARY_DIR}/generated")
add_compile_definitions(HFC_CONFIGURED_FONTS)
add_executable(hyperframes-cpp-native main.cpp)
add_executable(gptframes-word-align tools/native_word_align.cpp)
if (HFC_HTML_LAYOUT)
  if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/_sources/litehtml/CMakeLists.txt")
    message(FATAL_ERROR "litehtml source missing: run scripts/fetch-html-deps.sh")
  endif()
  set(BUILD_SHARED_LIBS OFF CACHE BOOL "Static litehtml" FORCE)
  set(LITEHTML_ENABLE_LINT OFF CACHE BOOL "" FORCE)
  set(LITEHTML_BUILD_TESTING OFF CACHE BOOL "" FORCE)
  add_subdirectory(third_party/_sources/litehtml ${CMAKE_BINARY_DIR}/thirdparty/litehtml EXCLUDE_FROM_ALL)
  target_link_libraries(hyperframes-cpp-native PRIVATE litehtml)
  target_compile_definitions(hyperframes-cpp-native PRIVATE HFC_WITH_LITEHTML)
  add_executable(hyperframes-html-preview html_layout_preview.cpp)
  target_link_libraries(hyperframes-html-preview PRIVATE litehtml)
endif()
option(HFC_THORVG "Enable native ThorVG SVG/Lottie layers" OFF)
if(HFC_THORVG)
  set(_TVG_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/_install/thorvg")
  find_path(HFC_THORVG_INCLUDE thorvg.h
    PATHS "${_TVG_ROOT}/include/thorvg-1" NO_DEFAULT_PATH)
  find_library(HFC_THORVG_LIBRARY NAMES thorvg-1
    PATHS "${_TVG_ROOT}/lib" "${_TVG_ROOT}/lib/${CMAKE_LIBRARY_ARCHITECTURE}" NO_DEFAULT_PATH)
  if(NOT HFC_THORVG_INCLUDE OR NOT HFC_THORVG_LIBRARY)
    message(FATAL_ERROR "ThorVG not installed. See docs/MOTION_STUDIO.md")
  endif()
  target_include_directories(hyperframes-cpp-native PRIVATE "${HFC_THORVG_INCLUDE}")
  target_link_libraries(hyperframes-cpp-native PRIVATE "${HFC_THORVG_LIBRARY}")
  target_compile_definitions(hyperframes-cpp-native PRIVATE HFC_WITH_THORVG)
endif()
option(HFC_NATIVE_MP4 "Build native H.264/AAC MP4 encoder" OFF)
if(HFC_NATIVE_MP4)
  foreach(dep minih264 minimp4 vo-aacenc)
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/_sources/${dep}")
      message(FATAL_ERROR "${dep} missing, run scripts/fetch-codecs.sh")
    endif()
  endforeach()
  set(CODEC_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/_sources")
  file(GLOB AAC_SRC "${CODEC_ROOT}/vo-aacenc/aacenc/src/*.c")
  add_library(hfc-native-codecs STATIC
    native_codecs/h264_impl.c
    native_codecs/mp4_impl.c
    "${CODEC_ROOT}/vo-aacenc/common/cmnMemory.c"
    "${CODEC_ROOT}/vo-aacenc/aacenc/basic_op/basicop2.c"
    "${CODEC_ROOT}/vo-aacenc/aacenc/basic_op/oper_32b.c"
    ${AAC_SRC})
  target_compile_definitions(hfc-native-codecs PRIVATE H264E_MAX_THREADS=0 __unused=)
  target_include_directories(hfc-native-codecs PUBLIC
    "${CODEC_ROOT}/minih264"
    "${CODEC_ROOT}/minimp4"
    "${CODEC_ROOT}/vo-aacenc"
    "${CODEC_ROOT}/vo-aacenc/aacenc/inc"
    "${CODEC_ROOT}/vo-aacenc/aacenc/basic_op"
    "${CODEC_ROOT}/vo-aacenc/common/include")
  target_compile_definitions(hyperframes-cpp-native PRIVATE HFC_WITH_NATIVE_MP4)
  target_link_libraries(hyperframes-cpp-native PRIVATE hfc-native-codecs)
  if(NOT WIN32)
    target_link_libraries(hyperframes-cpp-native PRIVATE m)
  endif()
endif()

option(HFC_BROWSER_CAPTURE "Enable installed-browser CDP capture mode" ON)

  endif()
endif()

# Keep all C/C++ runtime libraries within the executable; external browser
# execution is intentionally the only required runtime program for browser mode.
if(MINGW)
  target_link_options(hyperframes-cpp-native PRIVATE -static -static-libgcc -static-libstdc++)
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_link_options(hyperframes-cpp-native PRIVATE -static -static-libgcc -static-libstdc++)
endif()


option(HFC_MCP "Build the native C++ MCP service" ON)
include(CTest)
if(HFC_MCP)
  find_package(Threads REQUIRED)
  add_library(gptframes-mcp-core INTERFACE)
  target_compile_features(gptframes-mcp-core INTERFACE cxx_std_17)
  target_include_directories(gptframes-mcp-core INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(gptframes-mcp-core INTERFACE Threads::Threads)
  if(WIN32)
    target_compile_definitions(gptframes-mcp-core INTERFACE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(gptframes-mcp-core INTERFACE winhttp ws2_32 crypt32 bcrypt)
  else()
    find_package(CURL REQUIRED)
    find_package(OpenSSL REQUIRED COMPONENTS Crypto)
    target_link_libraries(gptframes-mcp-core INTERFACE CURL::libcurl OpenSSL::Crypto)
  endif()
  add_executable(gptframes-mcp mcp/server.cpp)
  target_link_libraries(gptframes-mcp PRIVATE gptframes-mcp-core)
  target_compile_definitions(gptframes-mcp PRIVATE GPTFRAMES_DEFAULT_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
  if(MINGW)
    target_link_options(gptframes-mcp PRIVATE -static -static-libgcc -static-libstdc++)
  endif()
  if(BUILD_TESTING)
    add_executable(gptframes-mcp-test tests/mcp_test.cpp)
    target_link_libraries(gptframes-mcp-test PRIVATE gptframes-mcp-core)
    add_test(NAME mcp-native COMMAND gptframes-mcp-test $<TARGET_FILE:hyperframes-cpp-native> "${CMAKE_CURRENT_SOURCE_DIR}")
    set_tests_properties(mcp-native PROPERTIES TIMEOUT 60)
    add_executable(gptframes-render-http-test tests/render_http_test.cpp)
    target_link_libraries(gptframes-render-http-test PRIVATE gptframes-mcp-core)
    add_test(NAME render-private-http COMMAND gptframes-render-http-test $<TARGET_FILE:hyperframes-cpp-native> "${CMAKE_CURRENT_SOURCE_DIR}")
    set_tests_properties(render-private-http PROPERTIES TIMEOUT 60)
    add_executable(gptframes-render-jobs-test tests/render_jobs_test.cpp)
    target_link_libraries(gptframes-render-jobs-test PRIVATE gptframes-mcp-core)
    add_test(NAME render-queue COMMAND gptframes-render-jobs-test $<TARGET_FILE:hyperframes-cpp-native> "${CMAKE_CURRENT_SOURCE_DIR}")
    set_tests_properties(render-queue PROPERTIES TIMEOUT 60)
    add_executable(gptframes-mcp-http-test tests/mcp_http_test.cpp)
    target_link_libraries(gptframes-mcp-http-test PRIVATE gptframes-mcp-core)
    add_test(NAME mcp-http-stdio COMMAND gptframes-mcp-http-test $<TARGET_FILE:gptframes-mcp> "${CMAKE_CURRENT_SOURCE_DIR}")
    set_tests_properties(mcp-http-stdio PROPERTIES TIMEOUT 60)
  endif()
endif()
